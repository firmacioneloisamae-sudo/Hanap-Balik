/* ---------- Item cards and search ---------- */

function card(d) {
    return (
        '<article class="card">' +
            '<div class="thumb">' + I("box") + "</div>" +
            '<div style="display:flex;justify-content:space-between;gap:8px">' +
                "<h3>" + d[0] + "</h3>" +
                '<span class="badge ' + d[4] + '">' + d[1] + "</span>" +
            "</div>" +
            '<div class="meta">' +
                "<span>" + I("pin") + d[2] + "</span>" +
                "<span>" + I("cal") + d[3] + "</span>" +
            "</div>" +
            '<a class="btn out" style="width:100%" href="#claims">View details</a>' +
        "</article>"
    );
}

function items() {
    var keyword = ($("#kw") || { value: "" }).value.toLowerCase();
    var status = ($("#fs") || { value: "" }).value;
    var results = DATA.filter(function (d) {
        var matchesKeyword = !keyword ||
            (d[0] + " " + d[2]).toLowerCase().includes(keyword);
        var matchesStatus = !status || d[1] === status;
        return matchesKeyword && matchesStatus;
    });

    var resultsContainer = $("#sr");

    if (!resultsContainer) return;

    resultsContainer.innerHTML = results.length
        ? results.map(card).join("")
        : '<div class="card empty" style="grid-column:1/-1">' +
            '<b style="color:var(--tx)">No items match your search</b><br>' +
            "Try fewer keywords or clear the filters." +
          "</div>";
}

function renderRecentItems() {
    document.querySelectorAll(".items:not(#sr)").forEach(function (el) {
        el.innerHTML = DATA.slice(0, 3).map(card).join("");
    });
}

/* ---------- Admin claim verification ---------- */

function adm() {
    var queryInput = $("#aq");
    var statusInput = $("#as");
    var tableBody = $("#at");

    if (!queryInput || !statusInput || !tableBody) return;

    var query = queryInput.value.toLowerCase();
    var status = statusInput.value;

    var rows = CL.filter(function (c) {
        var matchesStatus = !status || c[3] === status;
        var matchesQuery = c.join(" ").toLowerCase().includes(query);
        return matchesStatus && matchesQuery;
    }).map(function (c) {
        var actions = c[3] === "Pending"
            ? '<button class="btn out" type="button" ' +
                'style="min-height:36px;padding:6px 12px" ' +
                'onclick="dlg(\'Approve claim ' + c[0] + '?\', ' +
                '\'The claimant will be notified and the item moves to release.\', ' +
                'function(){updateClaim(\'' + c[0] + '\', \'Approved\')})">' +
                "Approve</button> " +

              '<button class="btn dng" type="button" ' +
                'style="min-height:36px;padding:6px 12px" ' +
                'onclick="dlg(\'Reject claim ' + c[0] + '?\', ' +
                '\'The claimant will be asked for stronger evidence.\', ' +
                'function(){updateClaim(\'' + c[0] + '\', \'Rejected\')})">' +
                "Reject</button>"
            : '<span class="mu sm">Reviewed</span>';

        return (
            "<tr>" +
                "<td>" + c[0] + "</td>" +
                "<td>" + c[1] + "</td>" +
                "<td>" + c[2] + "</td>" +
                '<td><span class="badge ' + BC[c[3]] + '">' +
                    c[3] + "</span></td>" +
                "<td>" + actions + "</td>" +
            "</tr>"
        );
    });

    tableBody.innerHTML = rows.join("") ||
        '<tr><td colspan="5" class="empty">No claims match.</td></tr>';
}

function updateClaim(claimId, newStatus) {
    var claim = CL.find(function (c) {
        return c[0] === claimId;
    });

    if (!claim) return;

    claim[3] = newStatus;
    adm();
    toast("Claim " + newStatus.toLowerCase());
}

/* ---------- UI utilities ---------- */

function dlg(title, message, onConfirm) {
    var dialog = $("#dg");
    var titleElement = $("#dt");
    var messageElement = $("#dp");
    var confirmButton = $("#dok");

    if (!dialog || !titleElement || !messageElement || !confirmButton) {
        return;
    }

    titleElement.textContent = title;
    messageElement.textContent = message;

    confirmButton.onclick = function () {
        dialog.close();
        if (typeof onConfirm === "function") {
            onConfirm();
        }
    };

    if (typeof dialog.showModal === "function") {
        dialog.showModal();
    }
}

function toast(message) {
    var el = $("#toast");
    if (!el) return;

    el.textContent = message;
    el.classList.add("on");

    setTimeout(function () {
        el.classList.remove("on");
    }, 2400);
}

    input.type = input.type === "password" ? "text" : "password";
}

function theme() {
    var root = document.documentElement;
    var currentTheme = root.dataset.theme || "light";
    root.dataset.theme = currentTheme === "dark" ? "light" : "dark";
}

/* ---------- Form validation ---------- */

function bad(field, isValid) {
    if (!field) return false;

    field.classList.toggle("err", !isValid);

    var errorMessage = field.parentNode
        ? field.parentNode.querySelector(".er")
        : null;

    if (errorMessage) {
        errorMessage.classList.toggle("on", !isValid);
    }

    return isValid;
}

function val(form, rules) {
    var allValid = true;

    rules.forEach(function (rule) {
        var field = $("#" + rule[0]);

        if (!field || !bad(field, rule[1](field.value))) {
            allValid = false;
        }
    });

    return allValid;
}

var em = function (value) {
    return /^\S+@\S+\.\S+$/.test(value);
};

var ne = function (value) {
    return value.trim().length > 0;
};

/* ---------- Form handlers ---------- */

function setupForms() {
    var loginForm = $("#lf");

    if (loginForm) {
        loginForm.onsubmit = function (e) {
            e.preventDefault();

            var valid = val(null, [
                ["le", em],
                ["lp", function (v) {
                    return v.length >= 8;
                }]
            ]);

            if (valid) {
                var button = $("#lb");

                if (button) {
                    button.textContent = "Logging in…";
                    button.disabled = true;
                }

                setTimeout(function () {
                    if (button) {
                        button.textContent = "Log in";
                        button.disabled = false;
                    }
                    go("dashboard");
                }, 900);
            }
        };
    }

    var signupForm = $("#sf");

    if (signupForm) {
        signupForm.onsubmit = function (e) {
            e.preventDefault();

            var valid = val(null, [
                ["sn", ne],
                ["sc", function (v) {
                    return v.replace(/\D/g, "").length >= 10;
                }],
                ["se", em],
                ["sp", function (v) {
                    return v.length >= 8;
                }],
                ["sq", function (v) {
                    return v.length > 0 && v === ($("#sp") || {}).value;
                }],
                ["si", ne]
            ]);

            var terms = $("#sa");

            if (valid && terms && !terms.checked) {
                toast("Please accept the Terms to continue");
                valid = false;
            }

            if (valid) {
                toast("Account created");
                go("dashboard");
            }
        };
    }

    var reportForm = $("#rf");

    if (reportForm) {
        reportForm.onsubmit = function (e) {
            e.preventDefault();

            if (val(null, [
                ["rn", ne],
                ["rd", ne],
                ["rl", ne]
            ])) {
                toast("Report submitted");
                go("dashboard");
            }
        };
    }
}

/* ---------- Navigation ---------- */

function renderNav(page) {
    var navElement = $("#nav");
    if (!navElement) return;

    var links = NAV[page === "admin" ? "admin" : "user"];
    var firstMatch = links.findIndex(function (link) {
        return link[0] === page;
    });

    var linkHtml = links.map(function (link, index) {
        var cls = index === firstMatch ? "on" : "";

        return '<a href="#' + link[0] + '" class="' + cls + '">' +
            I(link[1]) + link[2] + "</a>";
    }).join("");

    navElement.innerHTML =
        '<a class="logo" href="#home"><span class="mark">' +
        I("search") + "</span>HanapBalik" +
        (page === "admin" ? " Admin" : "") + "</a>" +
        linkHtml +
        '<span class="gap"></span>' +
        '<a href="#login">' + I("out") + "Log out</a>" +
        '<button class="btn out" type="button" ' +
        'onclick="theme()" style="margin-top:8px">' +
        "Toggle theme</button>";

    navElement.classList.remove("open");
}

function showAppView(page) {
    var appPage = $("#p-app");
    var view = $("#v-" + page);
    var header = T[page];

    if (!appPage || !view || !header) return;

    appPage.classList.add("on");

    document.querySelectorAll(".v").forEach(function (element) {
        element.style.display = "none";
    });

    view.style.display = "block";

    var title = $("#ttl");
    var subtitle = $("#sub");
    var quickAction = $("#qa");

    if (title) title.textContent = header[0];
    if (subtitle) subtitle.textContent = header[1];
    if (quickAction) quickAction.innerHTML = header[2];

    renderNav(page);

    if (page === "search") items();
    if (page === "admin") adm();
}

function route() {
    var hash = (location.hash || "#home").slice(1);

    if (["how", "safety", "about"].includes(hash)) {
        go("home", hash);
        return;
    }

    if (!["home", "login", "signup"].concat(APP).includes(hash)) {
        hash = "home";
    }

    document.querySelectorAll(".page").forEach(function (page) {
        page.classList.remove("on");
    });

    if (APP.includes(hash)) {
        showAppView(hash);
    } else {
        var pageElement = $("#p-" + hash);
        if (pageElement) pageElement.classList.add("on");
    }

    window.scrollTo(0, 0);
}

function go(page, section) {
    if (section) {
        document.querySelectorAll(".page").forEach(function (element) {
            element.classList.remove("on");
        });

        var homePage = $("#p-home");
        if (homePage) homePage.classList.add("on");

        var target = document.getElementById(section);
        if (target) target.scrollIntoView();

        return;
    }

    location.hash = page;
}

/* ---------- Start the application ---------- */

function init() {
    renderRecentItems();
    setupForms();

    var navElement = $("#nav");

    if (navElement) {
        navElement.addEventListener("click", function () {
            navElement.classList.remove("open");
        });
    }

    window.addEventListener("hashchange", route);
    route();

    // Search and admin filters update as the user types or changes status.
    var keywordInput = $("#kw");
    var statusFilter = $("#fs");
    var adminQuery = $("#aq");
    var adminStatus = $("#as");

    if (keywordInput) keywordInput.addEventListener("input", items);
    if (statusFilter) statusFilter.addEventListener("change", items);
    if (adminQuery) adminQuery.addEventListener("input", adm);
    if (adminStatus) adminStatus.addEventListener("change", adm);
}

if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", init);
} else {
    init();
}

class User(UserMixin, db.Model):
    id = db.Column(db.Integer, primary_key=True)
    full_name = db.Column(db.String(150), nullable=False)
    contact_number = db.Column(db.String(30))
    email = db.Column(db.String(150), unique=True, nullable=False)
    password_hash = db.Column(db.String(256), nullable=False)

    id_type = db.Column(db.String(100))
    id_number = db.Column(db.String(100))
    id_file_path = db.Column(db.String(255))

    is_verified = db.Column(db.Boolean, default=False)
    is_admin = db.Column(db.Boolean, default=False)

    def set_password(self, password):
        self.password_hash = generate_password_hash(password)

    def check_password(self, password):
        return check_password_hash(self.password_hash, password)

    user = db.relationship("User", backref=db.backref("items", lazy=True))

-- HanapBalik database (MySQL / MariaDB)
-- Run once in MySQL Workbench or phpMyAdmin.
CREATE DATABASE IF NOT EXISTS HanapBalik
    CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE HanapBalik;

CREATE TABLE IF NOT EXISTS categories (
    category_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS locations (
    location_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    description VARCHAR(255)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS staff (
    staff_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    role VARCHAR(50) NOT NULL DEFAULT 'staff'
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS found_items (
    found_id INT AUTO_INCREMENT PRIMARY KEY,
    category_id INT NOT NULL,
    description TEXT NOT NULL,
    distinguishing VARCHAR(255),
    found_location INT NOT NULL,
    found_date DATE NOT NULL,
    stored_at VARCHAR(150),
    photo_url VARCHAR(255),
    logged_by INT NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'stored',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (category_id) REFERENCES categories(category_id),
    FOREIGN KEY (found_location) REFERENCES locations(location_id),
    FOREIGN KEY (logged_by) REFERENCES staff(staff_id)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS lost_reports (
    report_id INT AUTO_INCREMENT PRIMARY KEY,
    category_id INT NOT NULL,
    description TEXT NOT NULL,
    last_seen_loc INT NOT NULL,
    lost_date DATE NOT NULL,
    reporter_name VARCHAR(150) NOT NULL,
    reporter_email VARCHAR(150) NOT NULL,
    reporter_phone VARCHAR(30),
    status VARCHAR(20) NOT NULL DEFAULT 'report',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (category_id) REFERENCES categories(category_id),
    FOREIGN KEY (last_seen_loc) REFERENCES locations(location_id)
) ENGINE=InnoDB;
